### 1. 修复测量框架 (ptq1_oneclick.py)

**A/B/B/A 交错测量**：
- 每个候选的测试序列改为：baseline → candidate → candidate → baseline
- 计算前后两次 baseline 的差异百分比，超过 5% 则标记 DRIFT_DETECTED
- 使用配对比较：(candidate1 - baseline1) 和 (candidate2 - baseline2) 的平均值
- 如果漂移超过 10%，整轮测试标记为 UNSTABLE，建议重测

**热状态监控**：
- 每次测试前记录系统 thermalState（通过 API 查询或增加端点）
- 每次测试前执行 2 次短推理预热（256 token），丢弃结果
- 在报告中记录每次测试的热状态和前后 baseline 差异

**分离评估维度**：
- 为候选添加 optimization_target 属性：DECODE_FOCUSED / PREFILL_FOCUSED / BALANCED
- DECODE_FOCUSED 候选（如 PTQ1_MV_R5）：
  - 主要看 decode token/s 和单 token TTFT
  - prefill 允许 ±10% 波动，不作为淘汰条件
  - 门槛：decode 提升 ≥3% 且稳定
- PREFILL_FOCUSED 候选：
  - 主要看 prefill 时间
  - decode 允许下降 ≤10%
  - 门槛：prefill 提升 ≥5%
- BALANCED 候选保持原有严格标准（prefill ≥15%、TTFT ≥10%、decode 下降 ≤5%）

**分级门槛**：
- SIGNIFICANT：达到原标准（15%/10%）
- MODERATE：5-15% 提升，标注为"中等收益，建议进一步验证"
- MINOR：2-5% 提升且重复测试稳定，标注为"微小但可重复收益"
- NO_BENEFIT：<2% 或不稳定

**改进报告**：
- 输出每个候选的置信区间（mean ± std）
- 明确标注：SHORT_SCREEN_NO_BENEFIT（未进入重复比较）vs CONFIRMED_NO_BENEFIT（重复比较确认无收益）
- 记录实际场景覆盖：M=1 命中次数、M>1 命中次数、prefill/decode 占比
- 基线漂移记录：每轮前后 baseline 差异百分比

```swift
import Foundation
import UIKit
import Metal

class CrashDiagnostics {
    static let shared = CrashDiagnostics()
    
    private var breadcrumbs: [Breadcrumb] = []
    private var operationTimeline: [TimelineEvent] = []
    private var recentErrors: [DiagnosticError] = []
    private let maxBreadcrumbs = 100
    private let maxTimelineEvents = 200
    
    // 操作追踪
    func recordBreadcrumb(_ action: String, metadata: [String: Any] = [:]) {
        let crumb = Breadcrumb(
            timestamp: Date(),
            action: action,
            metadata: metadata
        )
        breadcrumbs.append(crumb)
        if breadcrumbs.count > maxBreadcrumbs {
            breadcrumbs.removeFirst()
        }
    }
    
    func recordTimelineEvent(_ event: String, details: [String: Any] = [:]) {
        let timelineEvent = TimelineEvent(
            timestamp: Date(),
            event: event,
            details: details
        )
        operationTimeline.append(timelineEvent)
        if operationTimeline.count > maxTimelineEvents {
            operationTimeline.removeFirst()
        }
    }
    
    func recordError(severity: ErrorSeverity, type: String, message: String) {
        let error = DiagnosticError(
            timestamp: Date(),
            severity: severity,
            type: type,
            message: message
        )
        recentErrors.append(error)
        if recentErrors.count > 50 {
            recentErrors.removeFirst()
        }
    }
    
    // 生成完整崩溃报告
    func generateCrashReport(
        exception: NSException?,
        signal: Int32?,
        additionalContext: [String: Any] = [:]
    ) -> CrashReport {
        let crashId = generateCrashId()
        
        let report = CrashReport(
            crashId: crashId,
            timestamp: Date(),
            appVersion: getAppVersion(),
            executiveSummary: generateExecutiveSummary(exception: exception, signal: signal),
            deviceInfo: collectDeviceInfo(),
            appState: collectAppState(),
            stackTrace: collectStackTrace(exception: exception),
            memoryState: collectMemoryState(),
            operationTimeline: operationTimeline,
            breadcrumbs: breadcrumbs,
            performanceSnapshot: collectPerformanceSnapshot(),
            kernelStatistics: collectKernelStatistics(),
            configuration: collectConfiguration(),
            recentErrors: recentErrors,
            diagnosticFiles: collectDiagnosticFiles(),
            aiAnalysis: generateAIAnalysis()
        )
        
        // 保存报告
        saveCrashReport(report)
        
        return report
    }
    
    // 收集设备信息
    private func collectDeviceInfo() -> DeviceInfo {
        let device = UIDevice.current
        return DeviceInfo(
            model: getDeviceModel(),
            marketingName: getMarketingName(),
            osVersion: "\(device.systemName) \(device.systemVersion)",
            gpu: getGPUName(),
            totalMemoryGB: ProcessInfo.processInfo.physicalMemory / 1_000_000_000,
            availableMemoryMB: getAvailableMemory(),
            diskSpaceGB: getAvailableDiskSpace(),
            batteryLevel: UIDevice.current.batteryLevel,
            thermalState: getThermalStateName(),
            lowPowerMode: ProcessInfo.processInfo.isLowPowerModeEnabled
        )
    }
    
    // 收集应用状态
    private func collectAppState() -> AppState {
        // 从 BonsaiEngine 和 APIServer 获取当前状态
        return AppState(
            apiServerRunning: APIServer.shared.isRunning,
            apiUptimeSeconds: APIServer.shared.uptime,
            activeSessions: APIServer.shared.activeSessions,
            currentCandidate: BonsaiEngine.shared.currentCandidate?.displayName ?? "NONE",
            modelLoaded: BonsaiEngine.shared.currentModelName ?? "NONE",
            modelSizeGB: BonsaiEngine.shared.modelSizeBytes / 1_000_000_000,
            contextSize: BonsaiEngine.shared.contextSize,
            gpuLayers: BonsaiEngine.shared.gpuLayers,
            currentOperation: BonsaiEngine.shared.currentOperation ?? "IDLE",
            operationStartedAt: BonsaiEngine.shared.operationStartTime,
            tokensGenerated: BonsaiEngine.shared.tokensGenerated,
            lastSuccessfulToken: BonsaiEngine.shared.lastSuccessfulToken,
            screenKeepOnEnabled: UserDefaults.standard.bool(forKey: "benchmark_keep_screen_on")
        )
    }
    
    // 收集内存状态
    private func collectMemoryState() -> MemoryState {
        var info = mach_task_basic_info()
        var count = mach_msg_type_number_t(MemoryLayout<mach_task_basic_info>.size)/4
        let kerr: kern_return_t = withUnsafeMutablePointer(to: &info) {
            $0.withMemoryRebound(to: integer_t.self, capacity: 1) {
                task_info(mach_task_self_,
                         task_flavor_t(MACH_TASK_BASIC_INFO),
                         $0,
                         &count)
            }
        }
        
        let appMemoryMB = kerr == KERN_SUCCESS ? info.resident_size / 1_048_576 : 0
        
        return MemoryState(
            appMemoryMB: Int(appMemoryMB),
            appMemoryPeakMB: Int(BonsaiEngine.shared.peakMemoryUsage / 1_048_576),
            gpuAllocatedMB: Int(getGPUAllocatedMemory() / 1_048_576),
            gpuUsedMB: Int(getGPUUsedMemory() / 1_048_576),
            systemPressure: getMemoryPressureName(),
            memoryWarningsReceived: BonsaiEngine.shared.memoryWarningCount,
            lastMemoryWarning: BonsaiEngine.shared.lastMemoryWarningTime,
            kvCacheSizeMB: Int(BonsaiEngine.shared.kvCacheSize / 1_048_576),
            modelWeightsMB: Int(BonsaiEngine.shared.modelWeightsSize / 1_048_576),
            intermediateBuffersMB: Int(BonsaiEngine.shared.intermediateBuffersSize / 1_048_576)
        )
    }
    
    // 生成 AI 分析提示
    private func generateAIAnalysis() -> AIAnalysis {
        var hypotheses: [String] = []
        var reproSteps: [String] = []
        var fixes: [String] = []
        
        // 基于收集的数据生成假设
        if recentErrors.contains(where: { $0.type == "MEMORY_PRESSURE" }) {
            hypotheses.append("Memory pressure reached critical level before crash")
            hypotheses.append("Possible GPU memory fragmentation")
            fixes.append("Implement proactive memory cleanup when pressure reaches WARNING")
            fixes.append("Add periodic Metal heap compaction during long sessions")
        }
        
        if let thermalState = operationTimeline.last(where: { $0.event == "THERMAL_STATE_CHANGED" }) {
            hypotheses.append("Thermal throttling may have contributed to crash")
            fixes.append("Monitor and handle thermal throttling more aggressively")
        }
        
        // 生成重现步骤
        reproSteps.append("Load model: \(BonsaiEngine.shared.currentModelName ?? "unknown")")
        reproSteps.append("Use candidate: \(BonsaiEngine.shared.currentCandidate?.displayName ?? "unknown")")
        if let opStart = operationTimeline.first(where: { $0.event == "INFERENCE_STARTED" }) {
            reproSteps.append("Run inference with prompt tokens: \(opStart.details["prompt_tokens"] ?? 0)")
        }
        reproSteps.append("Generate tokens continuously until crash")
        
        return AIAnalysis(
            rootCauseHypothesis: hypotheses,
            reproductionSteps: reproSteps,
            suggestedFixes: fixes
        )
    }
    
    // 保存崩溃报告
    private func saveCrashReport(_ report: CrashReport) {
        let encoder = JSONEncoder()
        encoder.dateEncodingStrategy = .iso8601
        encoder.outputFormatting = [.prettyPrinted, .sortedKeys]
        
        do {
            let jsonData = try encoder.encode(report)
            let crashDir = getCrashReportDirectory()
            let jsonPath = crashDir.appendingPathComponent("\(report.crashId).json")
            try jsonData.write(to: jsonPath)
            
            // 也生成 Markdown 版本
            let markdown = generateMarkdownReport(report)
            let mdPath = crashDir.appendingPathComponent("\(report.crashId).md")
            try markdown.write(to: mdPath, atomically: true, encoding: .utf8)
            
            print("Crash report saved: \(jsonPath.path)")
        } catch {
            print("Failed to save crash report: \(error)")
        }
    }
    
    // 生成人类可读的 Markdown 报告
    private func generateMarkdownReport(_ report: CrashReport) -> String {
        var md = """
        # Crash Report: \(report.crashId)
        
        **Time**: \(report.timestamp)  
        **App Version**: \(report.appVersion)  
        **Device**: \(report.deviceInfo.marketingName) (\(report.deviceInfo.model))  
        **OS**: \(report.deviceInfo.osVersion)
        
        ## Executive Summary
        
        - **Crash Type**: \(report.executiveSummary.crashType)
        - **Exception**: \(report.executiveSummary.exceptionName ?? "N/A")
        - **Crashed In**: \(report.executiveSummary.crashedInOperation)
        - **Likely Cause**: \(report.executiveSummary.likelyCause)
        
        > 🤖 **AI Analysis Hint**: \(report.executiveSummary.aiAnalysisHint)
        
        ## System State at Crash
        
        ### Device
        - Memory: \(report.memoryState.appMemoryMB) MB / \(report.deviceInfo.totalMemoryGB) GB
        - GPU Memory: \(report.memoryState.gpuUsedMB) / \(report.memoryState.gpuAllocatedMB) MB
        - Battery: \(Int(report.deviceInfo.batteryLevel * 100))%
        - Thermal: \(report.deviceInfo.thermalState)
        - Memory Pressure: \(report.memoryState.systemPressure)
        
        ### Application
        - Model: \(report.appState.modelLoaded)
        - Candidate: \(report.appState.currentCandidate)
        - Operation: \(report.appState.currentOperation)
        - Tokens Generated: \(report.appState.tokensGenerated)
        - API Uptime: \(report.appState.apiUptimeSeconds)s
        
        ## Timeline (Last 10 Events)
        
        """
        
        for event in report.operationTimeline.suffix(10) {
            md += "- **\(event.timestamp)**: \(event.event)\n"
            if !event.details.isEmpty {
                md += "  - Details: \(event.details)\n"
            }
        }
        
        md += """
        
        ## Recent Errors
        
        """
        
        for error in report.recentErrors {
            md += "- [\(error.severity)] **\(error.type)**: \(error.message) (at \(error.timestamp))\n"
        }
        
        md += """
        
        ## AI Analysis
        
        ### Root Cause Hypotheses
        
        """
        
        for (i, hypothesis) in report.aiAnalysis.rootCauseHypothesis.enumerated() {
            md += "\(i+1). \(hypothesis)\n"
        }
        
        md += """
        
        ### Suggested Fixes
        
        """
        
        for (i, fix) in report.aiAnalysis.suggestedFixes.enumerated() {
            md += "\(i+1). \(fix)\n"
        }
        
        md += """
        
        ---
        
        **Full JSON Report**: `\(report.crashId).json`
        """
        
        return md
    }
    
    // 生成唯一崩溃 ID
    private func generateCrashId() -> String {
        let formatter = DateFormatter()
        formatter.dateFormat = "yyyyMMdd_HHmmss"
        let dateStr = formatter.string(from: Date())
        let random = String(format: "%04X", arc4random_uniform(65536))
        return "CRASH_\(dateStr)_\(random)"
    }
    
    // 辅助函数
    private func getCrashReportDirectory() -> URL {
        let documentsDir = FileManager.default.urls(for: .documentDirectory, in: .userDomainMask)[0]
        let crashDir = documentsDir.appendingPathComponent("CrashReports")
        try? FileManager.default.createDirectory(at: crashDir, withIntermediateDirectories: true)
        return crashDir
    }
}

```swift
class HealthMonitor {
    static let shared = HealthMonitor()
    
    func startMonitoring() {
        Timer.scheduledTimer(withTimeInterval: 10.0, repeats: true) { _ in
            self.checkHealth()
        }
    }
    
    private func checkHealth() {
        // 检查内存压力
        let availableMB = getAvailableMemory()
        if availableMB < 500 {
            CrashDiagnostics.shared.recordError(
                severity: .WARNING,
                type: "LOW_MEMORY",
                message: "Available memory below 500 MB: \(availableMB) MB"
            )
            // 触发主动内存清理
            BonsaiEngine.shared.cleanupMemory()
        }
        
        // 检查性能降级
        let recentSpeed = BonsaiEngine.shared.getRecentDecodeSpeed()
        let baselineSpeed = BonsaiEngine.shared.baselineDecodeSpeed
        if recentSpeed < baselineSpeed * 0.7 {
            CrashDiagnostics.shared.recordError(
                severity: .WARNING,
                type: "PERFORMANCE_DEGRADATION",
                message: "Decode speed degraded by \(Int((1 - recentSpeed/baselineSpeed) * 100))%"
            )
        }
        
        // 检查 GPU 内存碎片
        let fragmentation = getGPUMemoryFragmentation()
        if fragmentation > 0.5 {
            CrashDiagnostics.shared.recordError(
                severity: .WARNING,
                type: "GPU_MEMORY_FRAGMENTATION",
                message: "GPU memory fragmentation at \(Int(fragmentation * 100))%"
            )
        }
        
        // 检查热状态
        if ProcessInfo.processInfo.thermalState == .serious || 
           ProcessInfo.processInfo.thermalState == .critical {
            CrashDiagnostics.shared.recordError(
                severity: .WARNING,
                type: "THERMAL_THROTTLING",
                message: "Device in \(getThermalStateName()) thermal state"
            )
        }
    }
}

## 崩溃报告使用指南

### 如何查看崩溃报告

1. **通过 API**：
   ```bash
   # 列出所有崩溃
   curl http://iPad-IP:8080/debug/crashes
   
   # 获取特定崩溃的 JSON 详情
   curl http://iPad-IP:8080/debug/crashes/CRASH_20250115_143052_A7F3
   
   # 获取人类可读的 Markdown 报告
   curl http://iPad-IP:8080/debug/crashes/CRASH_20250115_143052_A7F3/markdown
   ```

2. **通过文件浏览器**（如 iTunes 文件共享或 iMazing）：
   - 路径：`Documents/CrashReports/`
   - JSON 文件：完整的结构化数据
   - Markdown 文件：人类可读的摘要

### AI 分析崩溃报告的步骤

1. **读取 executive_summary**：快速了解崩溃类型和可能原因
2. **检查 operation_timeline**：追踪崩溃前的事件序列
3. **分析 memory_state**：确认是否是内存相关问题
4. **查看 breadcrumbs**：了解用户操作流程
5. **阅读 ai_analysis**：查看自动生成的假设和建议
6. **验证 recent_errors**：识别预警信号是否被忽略

### 常见崩溃模式和修复

报告中 `ai_analysis.root_cause_hypothesis` 和 `suggested_fixes` 会基于以下模式自动生成建议：

- **内存压力崩溃**：检查 memory_warnings_received，建议主动清理
- **GPU 内存碎片**：检查 metal_heap_fragmentation，建议周期性压缩
- **热节流崩溃**：检查 thermal_state 变化，建议降低负载
- **长时间推理崩溃**：检查 tokens_generated，建议添加安全上限
